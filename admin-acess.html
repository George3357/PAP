// Admin Access Helper document.addEventListener('DOMContentLoaded', function() { // Add admin access button to footer for easy access during development const footerBottom = document.querySelector('.footer-bottom'); if (footerBottom) { const adminAccessDiv = document.createElement('div'); adminAccessDiv.style.marginTop = '20px'; adminAccessDiv.style.textAlign = 'center'; adminAccessDiv.innerHTML = `

Acesso Rápido ao Painel Admin

Use as credenciais: admin@cardealz.com / admin123

`; footerBottom.appendChild(adminAccessDiv); // Add event listener to the quick admin login button const quickAdminLoginBtn = document.getElementById('quick-admin-login'); if (quickAdminLoginBtn) { quickAdminLoginBtn.addEventListener('click', function() { // Create admin user if it doesn't exist const users = JSON.parse(localStorage.getItem('users')) || []; const adminExists = users.some(user => user.role === 'admin'); if (!adminExists) { // Create admin user const adminUser = { id: Date.now(), name: 'Administrador', email: 'admin@cardealz.com', password: 'admin123', role: 'admin', createdAt: new Date().toISOString() }; // Add admin to users array users.push(adminUser); // Save users to localStorage localStorage.setItem('users', JSON.stringify(users)); } // Log in as admin const adminUser = users.find(user => user.role === 'admin'); if (adminUser) { localStorage.setItem('currentUser', JSON.stringify(adminUser)); window.location.href = 'admin.html'; } }); } } });
